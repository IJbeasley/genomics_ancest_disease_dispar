# phecode to ICD10 mapping from https://wei-lab.app.vumc.org/phecode-data/phecode_international_version
phecodes <- fread(here::here("data/icd/phecode_international_version_unrolled.csv"))
n_studies <- nrow(disease_mapping)

icd10_desc_map  = 
phecodes |>
    mutate(trait_clean = tolower(iconv(ICD_DESCRIPTION, 
                                                       to = "UTF-8"))
           ) |>
    select(trait_clean,    
           icd10_code = ICD10) 

# filter missing strings
icd10_desc_map  = 
  icd10_desc_map |>
  filter(!is.na(trait_clean) & 
         trait_clean != "") |>
  filter(!is.na(icd10_code) & 
         icd10_code != "")

matched_icd10_desc =
  icd10_desc_map  |>
  group_by(trait_clean) |>
  summarise(icd10_code = str_flatten(unique(icd10_code), 
                                     collapse = ", ", 
                                     na.rm = T), 
            .groups = "drop") 
  
# match by cleaned DISEASE/TRAIT
matched_icd10_desc = 
  matched_icd10_desc |>
    mutate(icd10_code_match = "ICD Description") |>
    mutate(icd10_code_origin = "DISEASE/TRAIT")

disease_mapping = 
disease_mapping |>
rows_patch(matched_icd10_desc,
           unmatched = "ignore",
           by = "trait_clean")

umls_cuis_icd10 <-
  umls_data |>
  filter(SAB %in% c("ICD10", "ICD10CM")) |>
  select(CODE, CUI, code_type = SAB) |> 
  group_by(CUI) |>
  summarise(CODE = str_flatten(unique(CODE),
                               collapse = ", ",
                               na.rm = T),
            code_type = str_flatten(unique(code_type),
                               collapse = ", ",
                               na.rm = T),
            .groups = "drop"
            ) |>
  rename(icd10_code = CODE) 

umls_icd10 =
  umls_data |>
  left_join(umls_cuis_icd10,
