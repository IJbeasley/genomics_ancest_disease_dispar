  encoding = "UTF-8"          # correct — the file really is UTF-8
)

# Repair malformed bytes, fold diacritics, and convert to ASCII.  This is important because the PheCode mapping file contains some non-ASCII characters (e.g., "é" in "Meniere disease") that are not present in the GWAS Catalog disease terms.  If we don't do this, we won't be able to match these terms.
ascii_key <- function(x) {
  x |>
    stringi::stri_enc_toutf8(validate = TRUE) |>   # malformed bytes -> U+FFFD
    stringi::stri_trans_nfc() |>                   # é as e+U+0301 folds like é as U+00E9
    stringi::stri_trans_general("Latin-ASCII") |>  # é->e  ö->o  ç->c  ß->ss
    stringi::stri_replace_all_regex("[\u2010-\u2015]", "-") |>  # en/em dashes
    stringi::stri_replace_all_regex("[^\\p{ASCII}]", "") |>     # U+FFFD, β, × ...
    stringi::stri_trim_both() 
}

# H81.0 cannot be recovered from its bytes — repair it explicitly.
phecodes <- phecodes |>
  mutate(ICD_DESCRIPTION = if_else(ICD10 == "H81.0", "Meniere disease", ICD_DESCRIPTION)) |>
  mutate(
         ICD_DESCRIPTION        = ascii_key(ICD_DESCRIPTION),
         Phenotype   = ascii_key(Phenotype))
