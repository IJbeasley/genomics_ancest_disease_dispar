set_file_name <- function(date = Sys.getenv("RUN_TAG"),
                          # format(Sys.Date(), 
                          #        "%b %d %Y"), # get today's date in format "Nov 4 2024"
                          file_type,
                          paper_set = "training",

  file_name <-
  paste0("output/doccano/",
         paper_set, "_",
         article_section, "_",
         stringr::str_replace_all(date, " ", "_"),

  return(file_name)

# read/load relevant sentences from a section (abstracts, methods, results, intros) for a set of PMIDs
# won't work for titles 
read_section_sentences <- function(section,
                                   pmids) {
  
  
    stopifnot(section %in% c("abstracts",
                           "methods",
                           "results",
                           "intros")
            )

  sentences_path <- here::here(paste0(
   "output", "/",
   section, "/",
   section, "_sentences", "/"
  ))
  
  stopifnot(dir.exists(sentences_path))

  sentences_files <- list.files(sentences_path,
                                pattern = "*.json",
                                full.names = FALSE
  ) |>
    sort()

  sentences_pmids <-
  stringr::str_extract(sentences_files,
                           pattern = "^\\d+(?=_)")

  sentences_files <- sentences_files[sentences_pmids %in% pmids]

  sentences <- sapply(sentences_files,
                    function(file) {
                      json_data <- fromJSON(here::here(paste0(sentences_path,
                                                              file)))

                      sentences_lines <- json_data[json_data != ""]

                    }
  )

  return(sentences)

}

get_pubmed_titles <- function(pmids, 
                                 batch_size = 200, 
                                 verbose = TRUE) {
  
  n <- length(pmids)
  
  titles <- setNames(rep("MISSING", 
                            n), 
                        pmids
                        )  # initialize result
  
  # Split PMIDs into batches
  batches <- split(pmids, 
                   ceiling(seq_along(pmids)/batch_size)
                   )
  
  for(i in seq_along(batches)) {
    
    batch_pmids <- batches[[i]]
    
    if(verbose) message(sprintf("Fetching batch %d of %d (%d PMIDs)...", 
                                i, 
                                length(batches), 
                                length(batch_pmids)
                                )
                        )
    
    # Fetch XML
    xml_data <- entrez_fetch(db = "pubmed", 
                             id = paste(batch_pmids, 
                                        collapse = ","), 
                             rettype = "xml", 
                             parsed = FALSE
                             )
    
    doc <- read_xml(xml_data)
    articles <- xml_find_all(doc, ".//PubmedArticle")
    
    for(article in articles) {
      
      pmid_node <- xml_find_first(article, 
                                  ".//PMID")
      
      pmid <- xml_text(pmid_node)
      
      
      title_nodes <- xml_find_all(article, 
                                     ".//MedlineCitation/Article/ArticleTitle")
      
      if(length(title_nodes) > 0) {
        titles[pmid] <- paste(xml_text(title_nodes), 
                                 collapse = " ")
      }
    }
  }
  
  return(titles)
}


section_to_jsonl <- function(section,
                             pmids,
                             paper_set = "training",
                             run_tag = Sys.getenv("RUN_TAG")) {

  stopifnot(section %in% c("title",
                           "abstracts",
                           "methods",
                           "results",
                           "intros")
            )

  pmids <- unique(as.character(pmids))

  sentences <- if (section == "title") {
    get_pubmed_titles(pmids)
  } else {
    read_section_sentences(section, pmids)
  }

  json_list_sentences <-
    purrr::imap(sentences,
                function(sentences,
                         file_name) {
                  
                  if(section == "title") {
                     
                     df <- data.frame(source_file = "title",
                                      text = sentences,
                                      pubmed_id = file_name)
                  } else {

                     df <- data.frame(source_file = file_name,
                                      text = sentences)
                     
                     df <- mutate(df,
                               pubmed_id = stringr::str_extract(source_file,
                                                                pattern = "^\\d+(?=_)"
                                                                )
                               )
                  }
                  
                  df  <- mutate(df,
                                label = list(list()))

                  return(df)
                }
    )

  json_file = here::here(set_file_name(file_type = "json",
                                       paper_set = paper_set,
                                       article_section = section))


  jsonl_file = here::here(set_file_name(file_type = "jsonl",
                                        paper_set = paper_set,
                                        article_section = section))
  
  writeLines(toJSON(json_list_sentences,
                    auto_unbox = TRUE,
                    pretty = TRUE),
             json_file)

  json_data <- fromJSON(json_file,
                        simplifyVector = FALSE)

  json_rows <- unlist(json_data,
                      recursive = FALSE)

  # Open connection to JSONL file
  con <- file(jsonl_file, "w")

  # Loop over each element (object) and write as one line
  for (i in seq_along(json_rows)) {
    writeLines(toJSON(json_rows[[i]],
                      auto_unbox = TRUE),
               con)
  }

  close(con)
  
  print(paste0("JSONL file created for section: ", 
               section, " at: ", 
               fs::path_rel(jsonl_file,
                            start = here::here())
  ))

}

Combine training examples into a single jsonl file

Concatenate training examples, and deduplicate based on pubmed_id and text, keeping the first occurrence of each unique combination, and reassigning ids.)

deduplicate is needed because in many cases where methods, results and introduction sections can’t be identified from the xml file, the whole text is used as a fallback, which can lead to duplicate sentences across sections.


cd output/doccano

# combine training examples into a single jsonl file
# create new sentence id numbers 
jq -c  '.' \
  training_title_${RUN_TAG}.jsonl \
  training_abstracts_${RUN_TAG}.jsonl \
  training_methods_${RUN_TAG}.jsonl \
  training_results_${RUN_TAG}.jsonl \
  training_intros_${RUN_TAG}.jsonl \
| jq -c -s 'to_entries | map(.value + {orig_id: .value.id, id: (.key + 1)}) | .[]' \
  > combined_training_${RUN_TAG}.jsonl

# add source_section field based on source_file 
# source_section identifies the section of the article from which the sentence was extracted, e.g. title, abstract, methods, results, introduction, or full text (unknown section)
jq '
  . + {
    source_section:
      (if (.source_file | test("main"; "i")) then "full text (unknown section)"
       elif (.source_file | test("methods"; "i")) then "methods"
       elif (.source_file | test("results"; "i")) then "results"
       elif (.source_file | test("abstract"; "i")) then "abstract"
       elif (.source_file | test("title"; "i")) then "title"
       elif (.source_file | test("intro"; "i")) then "introduction"
       else null
       end)
  }
' combined_training_${RUN_TAG}.jsonl > combined_training_edit_${RUN_TAG}.jsonl
  
# deduplicate combined training examples based on pubmed_id and text, keeping the first occurrence of each unique combination, and reassigning ids
jq -c -s '
  reduce .[] as $r ({seen: {}, out: []};
    ($r.pubmed_id + "\u0000" + $r.text) as $k
    | if .seen[$k] then . else .seen[$k] = true | .out += [$r] end)
  | .out
  | to_entries | map(.value + {orig_id: .value.id, id: (.key + 1)}) | .[]
' combined_training_edit_${RUN_TAG}.jsonl > combined_training_dedup_${RUN_TAG}.jsonl

# remove source_file and orig_id fields from combined_training_dedup.jsonl
jq -c 'del(.source_file)' combined_training_dedup_${RUN_TAG}.jsonl > combined_training_dedup_simple_${RUN_TAG}.jsonl

jq -c 'del(.orig_id)' combined_training_dedup_simple_${RUN_TAG}.jsonl > combined_training_dedup_${RUN_TAG}.jsonl

rm combined_training_dedup_simple_${RUN_TAG}.jsonl
rm combined_training_${RUN_TAG}.jsonl
