## ---------------------------------------------------------------------------
## Recoding helper
##
## For each cell: try a whole-cell exact match first (this preserves any
## curated multi-term string), otherwise split on commas, map each part, and
## paste the unique results back together. Unmapped parts pass through
## unchanged so nothing is silently lost.
## ---------------------------------------------------------------------------

lookup_term  <- setNames(sero_map$disease_term, sero_map$term)
lookup_class <- setNames(sero_map$term_class,   sero_map$term)

recode_sero <- function(x, what = c("term", "class")) {
  what <- match.arg(what)
  tbl  <- if (what == "term") lookup_term else lookup_class

  purrr::map_chr(x, function(cell) {
    if (is.na(cell)) return(NA_character_)
    cell <- str_squish(str_to_lower(cell))

    if (cell %in% names(tbl)) return(tbl[[cell]])

    parts <- str_squish(str_split_1(cell, ",\\s*"))
    out <- purrr::map_chr(parts, function(p) {
      if (p %in% names(tbl)) {
        tbl[[p]]
      } else if (what == "term") {
        p                      # not a seropositivity term: leave alone
      } else {
        NA_character_          # no class for non-seropositivity terms
      }
    })
    out <- unique(out[!is.na(out)])
    if (length(out) == 0) NA_character_ else paste(out, collapse = ", ")
  })
}

tidy_terms <- function(x){
  x |> str_replace_all(",\\s*,", ",") |>
       str_replace_all("^\\s*,\\s*", "") |>
       str_replace_all("\\s*,\\s*$", "") |>
       str_squish() |>
       vapply(\(s) paste(unique(str_trim(str_split_1(s, ",\\s*"))), collapse = ", "), character(1))
}

gwas_study_info$collected_all_disease_terms = tidy_terms(gwas_study_info$collected_all_disease_terms)
