# remove title from sentences (as this is likely to contain cohort names but is not a sentence describing the cohort)

has_balanced_parens <- function(s) {
  stringr::str_count(s, fixed("(")) == stringr::str_count(s, fixed(")"))
}

get_title_sentences <- function(x, n_words_threshold = 5) {
                 
                 # get short sentences < 6 words
                 sentences <- x[stringr::str_count(x, " ") < n_words_threshold]
                 
                 # catch / inspect title sentences 
                 # upper_case <- "(?m)^\\s*[A-Z][A-Z\\s]+$"
                 
                 # get sentences that start with capital letter
                 sentences <- sentences[grepl("^(\\d+(\\.\\d+)*\\s+)?[A-Za-z]", sentences)]
                  #sentences[grepl("^[A-Z]", sentences)]
                 
                 # exclude sentences containing unmatched bracket as these are more likely to be complex sentences that are less likely to be titles or headings
                 #sentences <- sentences[!grepl("\\(|\\)", sentences)]
                 sentences <- sentences[has_balanced_parens(sentences)]
                 
                 # exclude sentences that end with a comma, as these are more likely to be incomplete sentences that are part of a larger sentence, and less likely to be titles or headings
                 sentences <- sentences[!grepl(",$|\\($", sentences)] 
                 
                 # exclude sentences that contain urls
                 sentences <- sentences[!grepl("www.|http", sentences)]
                 
                 return(sentences)
                 
}

get_non_title_sentences <- function(x) {
  
                title_sentences <- get_title_sentences(x)
                 
                keep_non_title_sentences <- x[!c(x %in% title_sentences)]
                 
                return(keep_non_title_sentences)
}

extract_cohort_sentences <- function(text_list, # list of texts to search through (e.g. abstracts, or full texts)
                                     terms, # terms to grep search for
                                     column_name = "COHORT",
                                     tokenize = FALSE,
                                     ignore_case) {
  
  #word boundary or preceded by ( [ ,
  terms_grep <- paste0("(?<=\\b|\\(|\\[)",
                            terms,
                            "(?=\\b|,|\\(|\\])")

    } else {

    }
    
    # For each sentence, find all matching cohort names
    lapply(seq_along(sentences), function(s) {
      
      sentence <- sentences[s]
      
      # Identify cohort names present in this sentence
      matched_cohorts <- terms[str_detect(sentence, 
                                          regex(terms_grep, 
                                                ignore_case = ignore_case)
                                          )
                               ]
      term_df <-
      data.frame(
        article_id = i,
        sentence_id = s,
        sentence = sentence,
        stringsAsFactors = FALSE
      )
      
      # Add column with matched term names (if any)
      term_df[[column_name]] <- ifelse(length(matched_cohorts) > 0,
                                       str_flatten(unique(matched_cohorts), 
                                                   collapse = "|", 
                                                   na.rm = T),
                                       ""
      )
      
      # Add column with binary label for whether any terms were matched
      term_df[[paste0("has_", tolower(column_name))]] <- ifelse(length(matched_cohorts) > 0, 
                                                                TRUE, 
                                                                FALSE)
      
      return(term_df)
      
    }) |> bind_rows()

Prepare data in Doccano JSON format

convert_to_doccano_json_sentence_level <- function(cohort_sentences_df) {
  doccano_list <- vector("list", nrow(cohort_sentences_df))

  for (i in seq_len(nrow(cohort_sentences_df))) {
    current_sentence <- cohort_sentences_df$sentence[i]
    matched_cohort   <- cohort_sentences_df$COHORT[i]
    pubmed_id_val    <- cohort_sentences_df$pubmed_id[i]
    date_val         <- cohort_sentences_df$date[i]
    cohort_val       <- cohort_sentences_df$cohort[i]
    country_val      <- cohort_sentences_df$country[i]

    if (is.na(matched_cohort) || matched_cohort == "") {
      entities <- list()
    } else {
      cohorts <- if (grepl("\\|", matched_cohort)) {
        unlist(strsplit(matched_cohort, split = "\\|"))
      } else {
        matched_cohort
      }

      matches <- do.call(rbind, lapply(cohorts, function(c) {
        stringr::str_locate_all(current_sentence,
                                stringr::fixed(c, ignore_case = TRUE))[[1]]
      }))

      matches[, "start"] <- matches[, "start"] - 1  # 0-based for Doccano

      entities <- lapply(seq_len(nrow(matches)), function(k) {
        list(start_offset = matches[k, "start"],
             end_offset   = matches[k, "end"],
             label        = "COHORT")
      })
    }

    doccano_list[[i]] <- list(
      text                  = current_sentence,
      pubmed_id             = pubmed_id_val,
      date                  = date_val,
      country               = country_val,
      gwas_cat_cohort_label = cohort_val,
      label                 = entities
    )
  }

  # Convert label entries to [start, end, label] vector form
  doccano_list <- lapply(doccano_list, function(x) {
    x$label <- lapply(x$label, function(l) c(l$start_offset, l$end_offset, l$label))
    x
  })

  doccano_list
}
