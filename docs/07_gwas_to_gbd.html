Calculate overlap between ICD-10 code sets

I.e. between non-fatal and cause of death ICD-10 codes for the same cause.

# Convert a base-36 character to its numeric value
base36_value <- function(char) {
  # Base-36 characters
  base36_chars <- c(as.character(0:9), LETTERS)
  
  match(toupper(char), base36_chars) - 1L
}

parse_icd10_code <- function(code) {
  
  # Clean input
  raw_code <- str_trim(code)
  
  # Split ICD-10 code into components:
  #   letter + digit + alphanumeric + optional decimal + remaining characters
  parts <- str_match(
    raw_code,
    "^([A-Za-z])([0-9])([0-9A-Za-z])\\.?([0-9A-Za-z]*)$"
  )
  
  # Check: are the ICD-10 Codes formatted properly such that code an be parsed into its components?
  if (is.na(parts[1, 1])) {
    stop("Unparseable ICD10 code: '", raw_code, "'")
  }
  
  # get the separate components of the code
  # e.g. for A07.8
  
  # letter part of code, in e.g. is A
  letter <- toupper(parts[1, 2])
  
  # first and second digit of code: 
  # in e.g.is 0
  first_digit <- parts[1, 3]
  # in e.g. is 7
  second_digit <- parts[1, 4]
  
  # decimal part of code
  # in e.g. is 8
  decimal_part <- parts[1, 5]
  
  
  
  # Convert characters after the decimal to a base-36 fraction
  fraction <- 0
  
  if (nchar(decimal_part) > 0) {
    decimal_chars <- str_split(decimal_part, "")[[1]]
    
    fraction <- sum(
      base36_value(decimal_chars) /
        36^seq_along(decimal_chars)
    )
  }
  
  parsed_icd10_code <-
  list(
    letter = letter,
    value = base36_value(first_digit) * 36 +
            base36_value(second_digit) +
            fraction,
    label = raw_code,
    bare = nchar(decimal_part) == 0
  )
  
  return(parsed_icd10_code)
  
}

parse_range <- function(range_string) {
  
  # Split range into start and end codes
  range_parts <- str_trim(str_split(range_string, "-")[[1]])
  
  # get parsed start and end codes
  start_code <- parse_icd10_code(range_parts[1])
  end_code <- parse_icd10_code(range_parts[length(range_parts)])
  
  # Start and end codes must belong to the same ICD-10 category
  if (start_code$letter != end_code$letter) {
    stop("Range spans letters: '", range_string, "'")
  }
  
  # A bare 3-character END code covers the entire category
  end_value <- if (end_code$bare) {
    end_code$value + 1 - 1e-9
  } else {
    end_code$value
  }
  
  parsed_range <-
  tibble(
    letter = start_code$letter,
    start_value = start_code$value,
    end_value = end_value,
    start_label = start_code$label,
    end_label = end_code$label
  )
  
  return(parsed_range)
  
}

# to be applied rowwise to data frame
parse_rangelist <- function(range_list) {
  
  range_list <- unlist(str_split(range_list, ", "))
  
  parsed_rangelist <- purrr::map(range_list, parse_range) |>
                      purrr::list_rbind()
  
  return(parsed_rangelist)
}

# after parsing ranges, convert back to a simplified string representation
# e.g. A07.8-A07.9, A08.0-A08.1, A08.3-A08.5, A09.0-A09.9
simplify_ranges <- function(range_data) {
  
  if (nrow(range_data) == 0) {
    return(NA_character_)
  }
  
  range_labels <- if_else(
    range_data$start_label == range_data$end_label,
    range_data$start_label,
    paste0(
      range_data$start_label,
      "-",
      range_data$end_label
    )
  )
  
  paste(range_labels, collapse = ", ")
}
# get the overlap between two parsed ICD-10 range lists
# i.e. a & b need to be output of parse_rangelist
intersect_ranges <- function(parsed_rangelist_a, 
                             parsed_rangelist_b) {
  
  output <- list()
  
  for (i in seq_len(nrow(parsed_rangelist_a))) {
    for (j in seq_len(nrow(parsed_rangelist_b))) {
      
      # Ranges must belong to the same ICD-10 category
      if (parsed_rangelist_a$letter[i] != parsed_rangelist_b$letter[j]) {
        next
      }
      
      # The intersection starts at the later start point
      if (parsed_rangelist_a$start_value[i] >= parsed_rangelist_b$start_value[j]) {
        
        start_value <- parsed_rangelist_a$start_value[i]
        start_label <- parsed_rangelist_a$start_label[i]
        
      } else {
        
        start_value <- parsed_rangelist_b$start_value[j]
        start_label <- parsed_rangelist_b$start_label[j]
      }
      
      # The intersection ends at the earlier end point
      if (parsed_rangelist_a$end_value[i] <= parsed_rangelist_b$end_value[j]) {
        
        end_value <- parsed_rangelist_a$end_value[i]
        end_label <- parsed_rangelist_a$end_label[i]
        
      } else {
        end_value <- parsed_rangelist_b$end_value[j]
        end_label <- parsed_rangelist_b$end_label[j]
      }
      
      # Keep only valid intersections
      if (start_value <= end_value) {
        output[[length(output) + 1]] <- tibble(
          letter = parsed_rangelist_a$letter[i],
          start_value = start_value,
          end_value = end_value,
          start_label = start_label,
          end_label = end_label
        )
      }
    }
  }
  
  overlap <- bind_rows(output)
  overlap <- simplify_ranges(overlap)
  
  return(overlap)
}

compare_icd10_ranges <- function(icd10_range_a, 
                                 icd10_range_b) {
  
 parsed_range_a <- parse_rangelist(icd10_range_a)
 parsed_range_b <- parse_rangelist(icd10_range_b)
                                            
 overlap <- intersect_ranges(parsed_range_a, 
                             parsed_range_b
                             ) 

 return(overlap)

}
