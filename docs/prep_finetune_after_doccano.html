build_label_list <- function(start, end, label) {
  keep <- !is.na(start) & !is.na(end) & !is.na(label)
  if (!any(keep)) return(list())
  pmap(list(start[keep], end[keep], label[keep]),
       \(s, e, l) list(s, e, l))
}
