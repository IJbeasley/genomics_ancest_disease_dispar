abstract_embeddings <- data.table::fread(here::here("output/clustering/abstracts_medcpt_embeddings.csv"))

PCA_test <- prcomp(abstract_embeddings, 
                   center = TRUE, 
                   scale. = TRUE) 

pca_table <-
data.frame(PC1 = PCA_test$x[,1], 
           PC2 = PCA_test$x[,2], 
           PC3 = PCA_test$x[,3], 
           PC4 = PCA_test$x[,4], 
           PC5 = PCA_test$x[,5], 
           PC6 = PCA_test$x[,6], 
           PC7 = PCA_test$x[,7], 
           PC8 = PCA_test$x[,8], 
           PC9 = PCA_test$x[,9], 
           PC10 = PCA_test$x[,10],
           PUBMED_ID = abstract_embeddings$PUBMED_ID) 


var_exp <- PCA_test$sdev^2 / sum(PCA_test$sdev^2)

data.frame(PC = paste0("PC", 1:length(var_exp)),
           PC_n = 1:length(var_exp),
           Variance_Explained = var_exp) |>
  filter(PC_n <= 10) |>
  mutate(PC_n = factor(PC_n, levels = 1:10)) |>
  ggplot(aes(x = PC_n, y = Variance_Explained)) +
  geom_col() +
  labs(x = "Principal Component", 
       y = "Proportion of Variance Explained", 
       title = "Variance explained by each principal component") + 
  theme_bw()
