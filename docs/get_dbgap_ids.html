# API Information / Documentation: https://www.ncbi.nlm.nih.gov/gap/sstr/swagger/
# API extracts this information, for e.g. https://www.ncbi.nlm.nih.gov/gap/sstr/report/phs000387.v1.p1
get_sample_size_df_api <- function(api_link = "https://www.ncbi.nlm.nih.gov/gap/sstr/api/v1/study",
                                   dbgap_id){
  
  url <- paste0(api_link, "/", 
                dbgap_id, 
                "/summary")
  
  response <- httr::GET(url, httr::timeout(60))
  Sys.sleep(0.5)  # be polite to the API
  
  if (httr::status_code(response) != 200) {
    warning("Failed to retrieve data for dbGaP ID ", dbgap_id, 
            ". Status code: ", response$status_code)
    return(NULL)
  }
  
  result_summary <- httr::content(response)
  
  consent_groups_df <- result_summary$study$consent_groups |> 
    dplyr::bind_rows() 
  
  return(consent_groups_df)
  
}
