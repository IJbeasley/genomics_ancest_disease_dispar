Introduction

This analysis checks how well the fine-tuned PubMedBERT model picks out cohort names from article text. It compares the model’s predicted entities with the hand-annotated (true) entities in the validation set in two ways:

  1. Entity-level, per paper: precision, recall and F1 for cohort names, based on the set of distinct entities found in each paper.
  2. Token-level confusion matrix: how often each true entity class is predicted as each class, across all entity types.

Entity-level performance (per paper)

Load validation predictions

# One row per sentence in the validation set, with true and predicted
# entities stored as ";"-separated strings
preds <- data.table::fread(
  here::here("output/text_mining_predictions/validation_entity_predictions.csv")
)

Reshape to one row per entity

ents <-
  preds |>
  # entity columns only, not the start / end positions
  select(pubmed_id, predicted_cohorts, true_cohorts) |>
  pivot_longer(
    cols = c("predicted_cohorts", "true_cohorts"),
    names_to = "col",
    values_to = "ent"
  ) |>
  filter(!is.na(ent)) |>
  # split ";"-separated entities into separate rows
  separate_rows(ent, sep = ";") |>
  mutate(
    # "true" or "predicted"
    source = str_extract(col, "^(true|predicted)"),
    # entity type, e.g. "predicted_cohorts" -> "cohort"
    type   = str_remove(str_remove(col, "^(true|predicted)_"), "s$"),
    # normalise case / whitespace so near-identical strings match
    ent    = str_squish(str_to_lower(ent))
  ) |>
  filter(ent != "") |>
  # count each entity once per paper
  distinct(pubmed_id, type, source, ent)

Match true and predicted entities

# For each paper and entity, flag whether it was annotated (true),
# predicted, or both, and classify it as a TP, FP or FN
matched <-
  ents |>
  mutate(present = TRUE) |>
  pivot_wider(names_from = source,
              values_from = present,
              values_fill = FALSE) |>
  mutate(tp = true & predicted,
         fp = !true & predicted,
         fn = true & !predicted)

Calculate precision, recall and F1

# Precision, recall and F1 for each paper, then the median across papers.
# Any existing grouping (e.g. by entity type) is kept, so a grouped input
# gives one row per group.
prf <- function(d) {
  d |>
    group_by(pubmed_id, .add = TRUE) |>
    summarise(across(c(tp, fp, fn), sum), .groups = "drop_last") |>
    mutate(precision = tp / (tp + fp),
           recall    = tp / (tp + fn),
           f1        = 2 * precision * recall / (precision + recall)) |>
    # 0/0 (e.g. no predictions in a paper) counts as 0
    mutate(across(c(precision, recall, f1), ~ replace_na(.x, 0))) |>
    summarise(across(c(precision, recall, f1), ~ median(.x, na.rm = TRUE)))
}

Confusion matrix (all entity types)

Load the confusion matrix

# Rows are true classes, columns are predicted classes; "O" means
# "not an entity"
confusion_mat <- data.table::fread(
  here::here("output/text_mining_predictions/validation_entity_confusion_matrix.csv")
)

# The first column holds the true class labels
names(confusion_mat)[1] <- "true"

# Matrix version, with true classes as row names
confusion_mat_m <-
  tibble::column_to_rownames(confusion_mat, var = "true") |>
  as.matrix()

TP, FP, FN and TN for cohorts

true_positives  <- confusion_mat_m["COHORT", "COHORT"]
# true cohorts predicted as some other class
false_negatives <- sum(confusion_mat_m["COHORT", ]) - true_positives
# other classes predicted as cohorts
false_positives <- sum(confusion_mat_m[, "COHORT"]) - true_positives
# everything else
true_negatives  <- sum(confusion_mat_m) - true_positives -
  false_negatives - false_positives

tibble(true_positives, false_positives, false_negatives, true_negatives) |>
  mutate(precision = true_positives / (true_positives + false_positives),
         recall    = true_positives / (true_positives + false_negatives))
